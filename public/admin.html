API Endpoints

1. Send OTP

POST {BASE_URL}/api/v1/send-otp
Headers:
  x-api-key: YOUR_API_KEY
  Content-Type: application/json
Body:
  {"number": "03XXXXXXXXX"}
Response: {"ok":true,"number":"923XXXXXXXXX","state":"sent"}

2. Verify OTP

POST {BASE_URL}/api/v1/verify-otp
Headers:
  x-api-key: YOUR_API_KEY
  Content-Type: application/json
Body:
  {"number": "03XXXXXXXXX", "otp": "123456"}
Response: {"ok":true,"message":"Verified"}

3. Gateway Status

GET {BASE_URL}/api/v1/status
Headers:
  x-api-key: YOUR_API_KEY

Code Snippets

const OTP_BASE = 'YOUR_BASE_URL';
const OTP_KEY  = 'YOUR_API_KEY';

async function sendOTP(phone) {
  const res = await fetch(`${OTP_BASE}/api/v1/send-otp`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': OTP_KEY
    },
    body: JSON.stringify({ number: phone })
  });
  return res.json();
}

async function verifyOTP(phone, otp) {
  const res = await fetch(`${OTP_BASE}/api/v1/verify-otp`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'x-api-key': OTP_KEY
    },
    body: JSON.stringify({ number: phone, otp })
  });
  return res.json();
}

// Usage:
// const result = await sendOTP('03001234567');
// const verified = await verifyOTP('03001234567', '123456');
import requests

OTP_BASE = 'YOUR_BASE_URL'
OTP_KEY  = 'YOUR_API_KEY'

def send_otp(phone):
    r = requests.post(f'{OTP_BASE}/api/v1/send-otp',
        headers={'x-api-key': OTP_KEY},
        json={'number': phone})
    return r.json()

def verify_otp(phone, otp):
    r = requests.post(f'{OTP_BASE}/api/v1/verify-otp',
        headers={'x-api-key': OTP_KEY},
        json={'number': phone, 'otp': otp})
    return r.json()

# Usage:
# result = send_otp('03001234567')

<?php
define('OTP_BASE', 'YOUR_BASE_URL');
define('OTP_KEY',  'YOUR_API_KEY');

function sendOTP($phone) {
    $ch = curl_init(OTP_BASE . '/api/v1/send-otp');
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => [
            'Content-Type: application/json',
            'x-api-key: ' . OTP_KEY
        ],
        CURLOPT_POSTFIELDS => json_encode(['number' => $phone])
    ]);
    return json_decode(curl_exec($ch), true);
}

function verifyOTP($phone, $otp) {
    $ch = curl_init(OTP_BASE . '/api/v1/verify-otp');
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => [
            'Content-Type: application/json',
            'x-api-key: ' . OTP_KEY
        ],
        CURLOPT_POSTFIELDS => json_encode(['number' => $phone, 'otp' => $otp])
    ]);
    return json_decode(curl_exec($ch), true);
}
